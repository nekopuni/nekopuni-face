            <div class="flex justify-between items-center mb-3">
                <span id="zoom-title" class="text-xs font-bold text-pink-400"></span>
                <button onclick="closeZoomModal()" class="text-xs text-slate-400 bg-slate-800 px-2.5 py-1 rounded-lg">閉じる</button>
            </div>
            <img id="zoomed-img" src="" class="w-full aspect-3/4 object-cover rounded-xl border border-slate-700 shadow-lg">

    <script>
        window.appMemory = {
            logs: [],
            chats: []
        };

        const video = document.getElementById('video');
        const canvas = document.getElementById('canvas');
        const cameraLoading = document.getElementById('camera-loading');
        const diagnoseBtn = document.getElementById('diagnose-btn');
        const resultArea = document.getElementById('result-area');
        
        const ADMIN_PASSWORD = "admin123";
        let titleClickCount = 0;
        let titleClickTimer = null;
        let currentResultData = null;
        let latestLogId = null;
        let lastCapturedPhotoUrl = null;

        function showCustomAlert(title, message) {
            const modal = document.getElementById('custom-alert-modal');
            document.getElementById('custom-alert-title').innerText = title;
            document.getElementById('custom-alert-message').innerText = message;
            document.getElementById('custom-alert-buttons').innerHTML = `<button onclick="closeCustomAlert()" class="w-full bg-indigo-600 font-bold py-2.5 rounded-xl text-xs">OK</button>`;
            modal.classList.remove('hidden'); modal.classList.add('flex');
        }

        function showCustomConfirm(title, message, onYes) {
            const modal = document.getElementById('custom-alert-modal');
            document.getElementById('custom-alert-title').innerText = title;
            document.getElementById('custom-alert-message').innerText = message;
            document.getElementById('custom-alert-buttons').innerHTML = `
                <button onclick="closeCustomAlert()" class="w-1/2 bg-slate-800 py-2.5 rounded-xl text-xs">キャンセル</button>
                <button id="conf-yes" class="w-1/2 bg-pink-600 font-bold py-2.5 rounded-xl text-xs">実行</button>
            `;
            document.getElementById('conf-yes').onclick = () => { closeCustomAlert(); onYes(); };
            modal.classList.remove('hidden'); modal.classList.add('flex');
        }

        function closeCustomAlert() {
            document.getElementById('custom-alert-modal').classList.add('hidden');
            document.getElementById('custom-alert-modal').classList.remove('flex');
        }

        function openZoomModal(url, title, info) {
            document.getElementById('zoomed-img').src = url;
            document.getElementById('zoom-title').innerText = title;
            document.getElementById('zoom-info').innerText = info;
            document.getElementById('image-zoom-modal').classList.remove('hidden');
            document.getElementById('image-zoom-modal').classList.add('flex');
        }
        function closeZoomModal() {
            document.getElementById('image-zoom-modal').classList.add('hidden');
            document.getElementById('image-zoom-modal').classList.remove('flex');
        }

        window.addEventListener('DOMContentLoaded', async () => {
            try {
                const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 } } });
                video.srcObject = stream;
                cameraLoading.style.display = 'none';
                diagnoseBtn.disabled = false;
                diagnoseBtn.classList.remove('opacity-50', 'cursor-not-allowed');
            } catch (err) {
                cameraLoading.innerText = "カメラの起動に失敗しました";
            }
        });

        function handleSecretClick() {
            titleClickCount++;
            clearTimeout(titleClickTimer);
            if (titleClickCount >= 3) {
                titleClickCount = 0;
                document.getElementById('password-modal').classList.remove('hidden');
                document.getElementById('password-modal').classList.add('flex');
            } else {
                titleClickTimer = setTimeout(() => { titleClickCount = 0; }, 1000);
            }
        }
        window.addEventListener('keydown', (e) => {
            if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'm') {
                e.preventDefault();
                document.getElementById('password-modal').classList.remove('hidden');
                document.getElementById('password-modal').classList.add('flex');
            }
        });

        const types = [
            { title: "🐱 猫顔タイプ", comments: ["ミステリアスな目元に全人類がひれ伏すレベル。気まぐれに生きる姿が最高です。"] },
            { title: "🐶 犬顔タイプ", comments: ["忠犬ハチ公もびっくりの愛されオーラ。全方位から可愛がられる才能の持ち主。"] },
            { title: "🦊 キツネ顔タイプ", comments: ["シャープな輪郭が美しすぎる！知性派スマートオーラがビンビンに漂っています。"] }
        ];

        function startDiagnosis() {
            diagnoseBtn.innerText = "AI分析中...";
            diagnoseBtn.disabled = true;

            canvas.width = 150;
            canvas.height = 200;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
            lastCapturedPhotoUrl = canvas.toDataURL('image/jpeg', 0.4);

            setTimeout(() => {
                const typeObj = types[Math.floor(Math.random() * types.length)];
                const mote = 75, intel = 80, charisma = 70, funny = 85;
                const total = 78;
                const aiComment = typeObj.comments[0];

                currentResultData = { title: typeObj.title, total, mote, intel, charisma, funny, comment: aiComment, photo: lastCapturedPhotoUrl };
                latestLogId = 'log_' + Date.now();

                document.getElementById('result-title').innerText = typeObj.title;
                document.getElementById('res-total').innerText = total;
                document.getElementById('res-mote').innerText = mote;
                document.getElementById('res-intel').innerText = intel;
                document.getElementById('res-charisma').innerText = charisma;
                document.getElementById('res-funny').innerText = funny;
                document.getElementById('res-ai-comment').innerText = aiComment;
                document.getElementById('user-custom-name').value = '';

                diagnoseBtn.classList.add('hidden');
                resultArea.classList.remove('hidden');

                window.appMemory.logs.unshift({
                    id: latestLogId,
                    time: new Date().toLocaleTimeString(),
                    userName: "名無しさん",
                    photo: lastCapturedPhotoUrl,
                    title: typeObj.title,
                    total: total
                });
            }, 1000);
        }

        function saveUserCustomName() {
            const nameInput = document.getElementById('user-custom-name').value.trim();
            if (!latestLogId) return;
            const finalName = nameInput || "名無しさん";

            const target = window.appMemory.logs.find(l => l.id === latestLogId);
            if (target) {
                target.userName = finalName;
                showCustomAlert("保存", `名前を「${finalName}」に変更しました！`);
            }
        }

        function verifyPassword() {
            if (document.getElementById('admin-pass-input').value === ADMIN_PASSWORD) {
                closeAdminModal();
                document.getElementById('user-view').classList.add('hidden');
                document.getElementById('admin-view').classList.remove('hidden');
                loadAdminDashboard();
            } else {
                showCustomAlert("エラー", "パスワードが違います");
            }
        }
        function closeAdminModal() {
            document.getElementById('password-modal').classList.add('hidden');
            document.getElementById('password-modal').classList.remove('flex');
        }
        function closeAdmin() {
            document.getElementById('admin-view').classList.add('hidden');
            document.getElementById('user-view').classList.remove('hidden');
        }

        function loadAdminDashboard() {
            const logs = window.appMemory.logs;
            document.getElementById('data-count').innerText = logs.length;
            const tbody = document.getElementById('log-table-body');
            tbody.innerHTML = '';

            if (logs.length === 0) {
                tbody.innerHTML = `<tr><td colspan="4" class="p-6 text-center text-slate-500">まだ診断データがありません（先に診断を行ってください）</td></tr>`;
            } else {
                logs.forEach(log => {
                    const tr = document.createElement('tr');
                    tr.innerHTML = `
                        <td class="p-3 text-xs text-slate-400 whitespace-nowrap">${log.time}</td>
                        <td class="p-3">
                            <img src="${log.photo}" onclick="openZoomModal('${log.photo}', '${log.userName}', '${log.title}')" class="w-12 h-12 object-cover rounded-xl border border-slate-700 cursor-pointer shadow">
                        </td>
                        <td class="p-3">
                            <span class="font-bold text-white">${log.userName}</span><br>
                            <span class="text-xs text-pink-400">${log.title}</span><br>
                            <span class="text-xs font-bold text-indigo-300">偏差値: ${log.total}</span>
                        </td>
                        <td class="p-3 text-center">
                            <button onclick="deleteLog('${log.id}')" class="bg-red-600/20 text-red-400 text-xs px-2.5 py-1.5 rounded-lg border border-red-500/30">削除</button>
                        </td>
                    `;
                    tbody.appendChild(tr);
                });
            }
            loadAdminChat();
        }

        function deleteLog(id) {
            window.appMemory.logs = window.appMemory.logs.filter(l => l.id !== id);
            loadAdminDashboard();
        }

        function confirmClearLogs() {
            showCustomConfirm("一括削除", "すべてのデータを削除しますか？", () => {
                window.appMemory.logs = [];
                loadAdminDashboard();
            });
        }

        function resetCamera() {
            resultArea.classList.add('hidden');
            diagnoseBtn.classList.remove('hidden');
            diagnoseBtn.innerText = "一瞬で診断する！";
            diagnoseBtn.disabled = false;
            latestLogId = null;
        }

        function generateShareCard() {
            showCustomAlert("保存", "シェアカードが生成されました！");
        }

        function sendGroupChatMessage() {
            const input = document.getElementById('group-chat-input');
            const msg = input.value.trim();
            const name = document.getElementById('admin-name-input').value.trim() || "管理者1";
            if(!msg) return;

            window.appMemory.chats.push({ name, message: msg, time: new Date().toLocaleTimeString() });
            input.value = '';
            loadAdminChat();
        }

        function loadAdminChat() {
            const container = document.getElementById('admin-group-messages');
            container.innerHTML = '';
            if(window.appMemory.chats.length === 0) {
                container.innerHTML = `<p class="text-center text-slate-500 py-2">メッセージはありません</p>`;
                return;
            }
            window.appMemory.chats.forEach(c => {
                const div = document.createElement('div');
                div.className = "bg-slate-900 p-2.5 rounded-xl border border-slate-800";
                div.innerHTML = `<span class="font-bold text-indigo-300">${c.name}</span> <span class="text-[10px] text-slate-500">${c.time}</span><p class="text-slate-300">${c.message}</p>`;
                container.appendChild(div);
            });
        }
    </script>
</body>
